  - className: string (optional)

  - depth: number (required)

  - isSelected: boolean (optional)
  - onHover: (id) => void (optional)
  - onClick: (id) => void (optional)
  - onInjectMRF: (id) => void (optional)
  - onInjectMCTS: (id) => void (optional)
  - showQueryPreview: boolean (default: true)

================================================================================
TYPE DEFINITIONS
================================================================================

TaskNode Interface:
  - id: string (unique identifier)
  - threadId: string (parent thread)
  - parentId: string|undefined (parent task)
  - childrenIds: string[] (child tasks)
  - depth: number (nesting level)
  - stepType: 'query'|'research'|'verify'|'synthesize'|'execute'
  - name: string (display name)
  - query: string|undefined (query text)
  - status: 'pending'|'running'|'completed'|'failed'|'skipped'
  - progressPct: number (0-100)
  - elapsedTimeMs: number
  - tokensUsed: number
  - confidence: number (0.0-1.0)
  - dependsOn: string[] (dependencies)
  - blocks: string[] (blocked tasks)
  - mrfEligible: boolean
  - mctsEligible: boolean
  - injectionApplied: string|undefined (injection name)
