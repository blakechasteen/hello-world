Promptly Core API

The main Promptly class provides prompt management and execution.

Promptly.__init__(data_dir=None)

Promptly(data_dir: str = None) -> Promptly

Initialize a new Promptly instance.

data_dir str, optional
Directory for storing prompts. Defaults to ~/.promptly
from promptly import Promptly

p = Promptly()
p = Promptly(data_dir="/custom/path")

add(name, content, metadata=None, tags=None)

add(name: str, content: str, metadata: dict = None, tags: List[str] = None) -> Prompt

Add a new prompt to the system.

name str
Unique identifier for the prompt
content str
The prompt template text. Supports {variable} syntax
metadata dict, optional
Optional metadata (author, source, etc.)
tags List[str], optional
Tags for categorization and search
p.add(
    name="code-reviewer",
    content="Review this {language} code: {code}",
    tags=["code", "review"],
    metadata={"author": "team", "version": "1.0"}
)

get(name, version=None, branch="main")

get(name: str, version: int = None, branch: str = "main") -> Prompt

Retrieve a prompt by name.

name str
Prompt name
version int, optional
Specific version. Latest if not specified
branch str, optional
Branch name. Defaults to "main"
prompt = p.get("code-reviewer")
prompt_v2 = p.get("code-reviewer", version=2)
feature_prompt = p.get("code-reviewer", branch="feature/advanced")

list(branch="main", tags=None, limit=None)

list(branch: str = "main", tags: List[str] = None, limit: int = None) -> List[Prompt]

List all prompts, optionally filtered by tags.

all_prompts = p.list()
code_prompts = p.list(tags=["code"])
recent = p.list(limit=10)

search(query, limit=10)

search(query: str, limit: int = 10) -> List[Prompt]

Search prompts by name or content.

results = p.search("code review", limit=5)

delete(name), update(name, content)

delete(name: str) -> bool update(name: str, content: str) -> Prompt

Delete or update a prompt.

p.delete("old-prompt")
p.update("code-reviewer", content="New template...")

Recursive Loops API

RecursiveLoop(loop_type, max_iterations, quality_threshold)

RecursiveLoop( loop_type: LoopType, max_iterations: int = 5, quality_threshold: float = 0.9, convergence_threshold: float = 0.02, timeout_seconds: int = None ) -> RecursiveLoop

Create a recursive loop for iterative reasoning.

from promptly.recursive_loops import RecursiveLoop, LoopType

loop = RecursiveLoop(
    loop_type=LoopType.REFINE,
    max_iterations=5,
    quality_threshold=0.85
)

run(initial_prompt, llm_fn)

run(initial_prompt: str, llm_fn: Callable) -> LoopResult

Execute the loop with the given prompt.

result = loop.run(
    initial_prompt="Write an essay about {topic}",
    llm_fn=lambda p: call_llm(p)
)

print(f"Final output: {result.final_output}")
print(f"Iterations: {result.iterations}")
print(f"Quality: {result.final_quality}")

Loop Composition API

LoopComposer()

LoopComposer() -> LoopComposer

Create a loop composer for defining pipelines.

from promptly.loop_composition import LoopComposer

composer = LoopComposer()

parse(dsl_string)

parse(dsl_string: str) -> Pipeline

Parse a loop composition DSL string into a pipeline.

pipeline = composer.parse("""
LOOP research_pipeline
  INPUT: topic

  STEP research:
    - DECOMPOSE topic into subtopics
    - Research each subtopic

  STEP analyze:
    - ANALYZE findings

  OUTPUT: research_summary
END
""")

Tools & Utilities

EnhancedLLMJudge

from promptly.tools.llm_judge_enhanced import EnhancedLLMJudge

judge = EnhancedLLMJudge()

# Evaluate a response
score = judge.evaluate(
    query: str,
    response: str,
    criteria: List[str],
    rubric: str = "standard"  # novice, intermediate, expert, standard
) -> Score

# Compare responses
winner = judge.compare(
    query: str,
    responses: List[str],
    criteria: List[str]
) -> ComparisonResult

# Batch evaluation
results = judge.batch_evaluate(
    evaluations: List[Dict]
) -> List[Score]

ABTester

from promptly.tools.ab_testing import ABTester

tester = ABTester()

# Compare two prompts
results = tester.compare(
    prompt_a: str,
    prompt_b: str,
    test_cases: List[Dict],
    criteria: List[str] = None,
    num_samples: int = 1
) -> ComparisonResults

CostTracker

from promptly.tools.cost_tracker import CostTracker

tracker = CostTracker()

# Track execution with cost
result = tracker.execute_and_track(
    prompt: str,
    variables: Dict,
    model: str = "gpt-4"
) -> ExecutionResult

# Get statistics
stats = tracker.get_stats() -> Dict

Return Types

Prompt

class Prompt:
    name: str
    content: str
    version: int
    branch: str
    created_at: datetime
    updated_at: datetime
    tags: List[str]
    metadata: Dict[str, Any]
    author: str

Score

class Score:
    overall: float  # 0.0-1.0
    scores: Dict[str, float]  # criterion -> score
    feedback: List[str]
    timestamp: datetime

LoopResult

class LoopResult:
    final_output: str
    iterations: int
    final_quality: float
    quality_trajectory: List[float]
    scratchpad: Scratchpad
    execution_time_ms: float
