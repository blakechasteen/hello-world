Prompt Composition

Prompt composition is the foundation of Promptly. It lets you chain multiple prompts together and build complex reasoning flows.

Basic Operations

from promptly import Promptly

p = Promptly()

# Add a prompt
p.add(
    name="analyze-code",
    content="Analyze this Python code for performance issues: {code}"
)

# Get a prompt
prompt = p.get("analyze-code")

# List all prompts
all_prompts = p.list()

# Search prompts
results = p.search("performance", limit=10)

# Delete a prompt
p.delete("analyze-code")

# Update a prompt
p.update("analyze-code", content="New content")

Version Control

Promptly implements Git-style version control for prompts:

# Create a branch for experimentation
p.branch("feature/advanced-analysis")

# Commit changes
p.commit("Add exception handling check")

# View history
history = p.log("analyze-code", limit=10)

# View a specific version
old_version = p.get("analyze-code", version=2)

# Merge branches
p.checkout("main")
p.merge("feature/advanced-analysis")

# View diffs between versions
diff = p.diff("analyze-code", v1=1, v2=2)

Variables & Templating

Use curly braces for variable interpolation:

# Simple variables
template = "Analyze this {language} code: {code}"

# Nested templates
p.add("code-reviewer", """
Analyze this {language} code:
{code}

Focus on:
1. {criterion1}
2. {criterion2}
3. {criterion3}
""")

# Use them
result = p.execute("code-reviewer", {
    "language": "Python",
    "code": "def foo(): pass",
    "criterion1": "Performance",
    "criterion2": "Security",
    "criterion3": "Readability"
})

Recursive Loops

Recursive loops enable iterative reasoning and self-improvement. They're inspired by Hofstadter's concept of strange loops and self-reference.

6 Loop Types

Type Purpose Best For
REFINE Iterative quality improvement Writing, code quality, explanations
CRITIQUE Self-critique and feedback loop Analysis, reviews, recommendations
DECOMPOSE Break complex problems into parts Architecture design, planning
VERIFY Generate and verify correctness Code generation, problem solving
EXPLORE Multiple solutions, ranking Brainstorming, optimization
HOFSTADTER Meta-reasoning, self-reference Philosophy, introspection, complex reasoning

REFINE Example

from promptly.recursive_loops import RecursiveLoop, LoopType

loop = RecursiveLoop(
    loop_type=LoopType.REFINE,
    max_iterations=5,
    quality_threshold=0.90
)

result = loop.run(
    initial_prompt="Write a blog post about {topic}",
    llm_fn=lambda p: call_llm(p)
)

# Access results
print(f"Iterations: {result.iterations}")
print(f"Final quality: {result.final_quality}")
print(f"Improvement: {result.quality_trajectory}")

CRITIQUE Example

critique_loop = RecursiveLoop(
    loop_type=LoopType.CRITIQUE,
    max_iterations=3
)

result = critique_loop.run(
    initial_prompt="""
Generate a solution for {problem}.
Then critique your solution.
Then improve it based on the critique.
""",
    llm_fn=call_llm
)

Composing Loops

Chain multiple loops together for complex workflows:

from promptly.loop_composition import LoopComposer

composer = LoopComposer()

# Define a pipeline
pipeline = composer.parse("""
LOOP code_review_pipeline
  INPUT: code

  STEP decompose:
    - DECOMPOSE the code into components

  STEP analyze:
    - ANALYZE each component for issues

  STEP verify:
    - VERIFY no issues were missed

  STEP refine:
    - REFINE the analysis for clarity

  OUTPUT: comprehensive_review
END
""")

# Execute the pipeline
result = pipeline.execute({"code": "..."})
💡 Loop Composition DSL

The DSL (Domain Specific Language) makes it easy to define complex pipelines without code. See examples.html for more.

Configuration & Tuning

# Fine-tune loop behavior
loop = RecursiveLoop(
    loop_type=LoopType.REFINE,
    max_iterations=10,           # Stop after 10 iterations
    quality_threshold=0.85,      # Stop if quality reaches 0.85
    convergence_threshold=0.02,  # Stop if improvement < 2%
    timeout_seconds=120          # Stop after 2 minutes
)

# Track progress
for iteration in loop.run_interactive(...):
    print(f"Iteration {iteration.num}: quality={iteration.quality}")
    if iteration.should_stop:
        break

Skills System

Skills are reusable, shareable prompt templates. Promptly includes 13 professional templates.

Using Built-in Skills

from promptly.skill_templates_extended import SkillLibrary

library = SkillLibrary()

# List available skills
skills = library.list_skills()

# Use a skill
result = library.execute("code-reviewer", {
    "code": "def process_data(data): ...",
    "focus": "security"
})

# Get skill details
skill_info = library.get_skill_info("sql-optimizer")

Creating Custom Skills

# Create a new skill
my_skill = library.create_skill(
    name="performance-auditor",
    description="Audit code for performance bottlenecks",
    template="""
Audit this {language} code for performance issues:
{code}

Identify:
1. Algorithmic inefficiencies
2. Memory leaks
3. I/O bottlenecks
4. Caching opportunities

Provide specific recommendations for each issue.
""",
    loop_type="CRITIQUE",
    max_iterations=3
)

# Save skill
library.save_skill(my_skill)

# Share skill with team
library.share_skill("performance-auditor", team="backend-team")

Testing & Quality

LLM Judge

Evaluate prompt responses using multi-criteria scoring:

from promptly.tools.llm_judge_enhanced import EnhancedLLMJudge

judge = EnhancedLLMJudge()

# Evaluate a single response
score = judge.evaluate(
    query="Explain machine learning",
    response="Machine learning is a subset of AI that...",
    criteria=["accuracy", "clarity", "completeness", "engagement"],
    rubric="expert_level"  # novice, intermediate, expert
)

print(f"Overall: {score.overall:.2f}")
print(f"Accuracy: {score.scores['accuracy']:.2f}")
print(f"Breakdown: {score.feedback}")

# Batch evaluation
results = judge.batch_evaluate([
    {"query": "What is AI?", "response": "..."},
    {"query": "Explain ML", "response": "..."}
])

# Compare responses
winner = judge.compare(
    query="Explain quantum computing",
    responses=[response_a, response_b],
    criteria=["accuracy", "clarity"]
)

A/B Testing

Compare prompt variations objectively:

from promptly.tools.ab_testing import ABTester

tester = ABTester()

# Define test prompts
prompt_a = "Explain {topic} in simple terms"
prompt_b = "Provide a detailed {topic} overview with examples"

# Test cases
test_cases = [
    {"topic": "Machine Learning"},
    {"topic": "Neural Networks"},
    {"topic": "Deep Learning"},
    {"topic": "Natural Language Processing"}
]

# Run A/B test
results = tester.compare(
    prompt_a=prompt_a,
    prompt_b=prompt_b,
    test_cases=test_cases,
    criteria=["clarity", "completeness", "usefulness"],
    num_samples=3  # Run each test 3 times
)

# Analyze results
print(f"Winner: {results.winner}")
print(f"Confidence: {results.confidence:.2f}")
print(f"Effect size: {results.effect_size:.2f}")

# Get detailed breakdown
for test_case in results.breakdown:
    print(f"{test_case.topic}: A={test_case.score_a:.2f}, B={test_case.score_b:.2f}")

Advanced Features

HoloLoom Integration

Leverage HoloLoom's semantic memory for context-aware prompts:

from promptly.integrations.hololoom_bridge import HoloLoomBridge

bridge = HoloLoomBridge()

# Query memory for context
context = bridge.recall("What did we discuss about MCTS?", limit=5)

# Use context in prompts
prompt_with_context = f"""
Based on our previous discussion:
{context}

Now answer: {new_question}
"""

# Store interactions for learning
bridge.remember(
    query="What is Thompson Sampling?",
    answer="Thompson Sampling is a Bayesian approach...",
    feedback={"helpful": True, "accuracy": 0.95}
)

Cost Tracking

from promptly.tools.cost_tracker import CostTracker

tracker = CostTracker()

# Track prompt execution costs
result = tracker.execute_and_track(
    prompt="Analyze this code: {code}",
    variables={"code": "..."},
    model="gpt-4"
)

# Get analytics
stats = tracker.get_stats()
print(f"Total cost: ${stats['total_cost']:.2f}")
print(f"Avg cost per execution: ${stats['avg_cost']:.4f}")
print(f"Most expensive prompts: {stats['most_expensive']}")

Analytics & Insights

from promptly.tools.prompt_analytics import AnalyticsEngine

analytics = AnalyticsEngine()

# Get overall statistics
summary = analytics.get_summary()
print(f"Total executions: {summary['total_executions']}")
print(f"Success rate: {summary['success_rate']:.1%}")
print(f"Avg quality: {summary['avg_quality']:.2f}")

# Per-prompt statistics
stats = analytics.get_prompt_stats("analyze-code")
print(f"Executions: {stats['executions']}")
print(f"Avg latency: {stats['avg_latency_ms']:.0f}ms")
print(f"Quality trajectory: {stats['quality_trend']}")
