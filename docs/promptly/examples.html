1. Code Review Pipeline

Multi-Stage Code Review with Recursive Refinement
INTERMEDIATE Uses: Decompose + Critique loops

Automatically review code across multiple dimensions and iteratively improve the feedback.

from promptly.recursive_loops import RecursiveLoop, LoopType
from promptly import Promptly

p = Promptly()

# Create decompose loop to split code into components
decompose_loop = RecursiveLoop(
    loop_type=LoopType.DECOMPOSE,
    max_iterations=1
)

# Create critique loop for improvement
critique_loop = RecursiveLoop(
    loop_type=LoopType.CRITIQUE,
    max_iterations=3,
    quality_threshold=0.85
)

code_sample = """
def process_user_data(data):
    result = []
    for item in data:
        if item['active']:
            result.append(item['name'].upper())
    return result
"""

# Stage 1: Decompose code into components
decompose_result = decompose_loop.run(
    initial_prompt=f"""
Analyze this code and identify its components:
{code_sample}

Break it down into:
1. Input validation
2. Core logic
3. Output formatting
""",
    llm_fn=lambda p: call_llm(p)
)

# Stage 2: Critique and improve
critique_result = critique_loop.run(
    initial_prompt=f"""
Original Code:
{code_sample}

Component Analysis:
{decompose_result.final_output}

Now provide a comprehensive code review covering:
- Performance issues
- Security concerns
- Code clarity and style
- Suggested improvements
""",
    llm_fn=lambda p: call_llm(p)
)

print("Review:", critique_result.final_output)

2. SQL Query Optimizer

Optimize and Verify SQL Queries
BEGINNER Uses: Verify loop, LLM Judge

Optimize SQL queries and verify the optimizations don't break functionality.

from promptly.recursive_loops import RecursiveLoop, LoopType
from promptly.tools.llm_judge_enhanced import EnhancedLLMJudge

# Create verify loop
verify_loop = RecursiveLoop(
    loop_type=LoopType.VERIFY,
    max_iterations=2
)

original_query = """
SELECT * FROM orders o
JOIN customers c ON o.customer_id = c.id
WHERE c.country = 'US'
ORDER BY o.created_at DESC
"""

result = verify_loop.run(
    initial_prompt=f"""
Optimize this SQL query:
{original_query}

The optimization must:
1. Return identical results
2. Improve performance (add indexes, better joins, etc.)
3. Be compatible with PostgreSQL

Provide the optimized query and explain the optimizations.
""",
    llm_fn=lambda p: call_llm(p)
)

# Evaluate the optimization
judge = EnhancedLLMJudge()
score = judge.evaluate(
    query="Is this SQL optimization valid and safe?",
    response=result.final_output,
    criteria=["correctness", "performance", "compatibility"]
)

print(f"Optimization quality: {score.overall:.2f}")
print(f"Optimized query:\n{result.final_output}")

4. Test Case Generator

Generate Comprehensive Test Cases
BEGINNER Uses: Explore loop

Generate multiple test case approaches and select the best one.

from promptly.recursive_loops import RecursiveLoop, LoopType

explore_loop = RecursiveLoop(
    loop_type=LoopType.EXPLORE,
    max_iterations=3
)

function_code = """
def calculate_discount(price, quantity, membership):
    discount_rate = 0
    if membership == 'gold':
        discount_rate = 0.15
    elif membership == 'silver':
        discount_rate = 0.10
    if quantity >= 10:
        discount_rate += 0.05
    return price * (1 - discount_rate)
"""

result = explore_loop.run(
    initial_prompt=f"""
Generate comprehensive test cases for this function:

{function_code}

Explore 3 different testing approaches:
1. Happy path testing (normal usage)
2. Edge case testing (boundaries)
3. Error case testing (invalid inputs)

For each approach, list specific test cases with expected outputs.
Then recommend the best combination of test cases.
""",
    llm_fn=lambda p: call_llm(p)
)

print(result.final_output)

5. Security Audit

Security Vulnerability Assessment
INTERMEDIATE Uses: Multiple loops, HoloLoom context

Comprehensive security audit with memory of past vulnerabilities.

from promptly.recursive_loops import RecursiveLoop, LoopType
from promptly.integrations.hololoom_bridge import HoloLoomBridge

# Get context from previous security audits
bridge = HoloLoomBridge()
past_vulnerabilities = bridge.recall(
    "Security vulnerabilities found in authentication code",
    limit=3
)

# Run security audit loop
audit_loop = RecursiveLoop(
    loop_type=LoopType.CRITIQUE,
    max_iterations=2
)

code = """
def login_user(username, password):
    user = db.query(f"SELECT * FROM users WHERE username='{username}'")
    if user and user.password == password:
        session.user_id = user.id
        return True
    return False
"""

result = audit_loop.run(
    initial_prompt=f"""
Security Audit of Login Function

Code to audit:
{code}

Previous vulnerability patterns to check for:
{past_vulnerabilities}

Identify ALL security issues including:
- SQL injection risks
- Authentication bypass
- Password storage issues
- Session management problems
- Input validation gaps

For each issue: describe it, rate severity, and provide a fix.
""",
    llm_fn=lambda p: call_llm(p)
)

# Store findings for future audits
bridge.remember(
    query="Security audit of login function",
    answer=result.final_output,
    feedback={"severity": "critical"}
)

print(result.final_output)

6. Prompt A/B Testing

Compare Prompt Variations Objectively
BEGINNER Uses: A/B Testing tool

Test two prompt approaches and determine which performs better.

from promptly.tools.ab_testing import ABTester

tester = ABTester()

# Define two prompt approaches
prompt_a = """
Explain {topic} to a {audience} in {style}.
"""

prompt_b = """
I need an explanation of {topic}.
Target audience: {audience}
Style preference: {style}

Please provide:
1. Clear definition
2. Key concepts
3. Practical examples
4. Common misconceptions
"""

# Test cases
test_cases = [
    {
        "topic": "Machine Learning",
        "audience": "high school student",
        "style": "simple and engaging"
    },
    {
        "topic": "Quantum Computing",
        "audience": "software engineer",
        "style": "technical and detailed"
    },
    {
        "topic": "Blockchain",
        "audience": "business executive",
        "style": "business-focused"
    }
]

# Run A/B test
results = tester.compare(
    prompt_a=prompt_a,
    prompt_b=prompt_b,
    test_cases=test_cases,
    criteria=["clarity", "completeness", "engagement"],
    num_samples=2
)

print(f"Winner: {results.winner} prompt")
print(f"Confidence: {results.confidence:.1%}")
print(f"\nDetailed results:")
for test in results.breakdown:
    print(f"  {test.topic}: A={test.score_a:.2f}, B={test.score_b:.2f}")

11. HoloLoom Integration Example

Use HoloLoom Memory in Prompts
ADVANCED Uses: HoloLoom Bridge

Leverage persistent memory to make prompts context-aware.

from promptly.integrations.hololoom_bridge import HoloLoomBridge
from promptly import Promptly

bridge = HoloLoomBridge()
p = Promptly()

# Retrieve relevant context from memory
context = bridge.recall(
    "What are the main challenges in ML deployment?",
    limit=5
)

# Create a prompt that uses this context
p.add("ml-deployment-expert",
    f"""
Based on our previous discussions:
{context}

Now answer: {'{question}'}

Build on the insights we've already discussed,
adding new perspectives and solutions.
""")

# Use the prompt
result = p.execute("ml-deployment-expert", {
    "question": "How can we reduce model serving latency?"
})

# Store the result for future learning
bridge.remember(
    query="How can we reduce model serving latency?",
    answer=result,
    feedback={"helpful": True}
)

print(result)

12. Creating a Custom Skill

Build and Share Reusable Skills
ADVANCED Uses: Skill system

Create a reusable prompt skill for your team.

from promptly import Promptly

p = Promptly()

# Create a custom skill
p.add(
    name="performance-audit",
    content="""
Perform a comprehensive performance audit on this code:

{code}

Analyze:
1. Time Complexity - What's the Big O complexity?
2. Space Complexity - Memory usage analysis
3. Bottlenecks - What's slow and why?
4. Optimizations - Specific improvements
5. Trade-offs - What are the alternatives?

Provide:
- Current performance metrics (estimated)
- Optimized version with improvements
- Benchmark comparison
- Implementation difficulty (1-10)
""",
    tags=["performance", "code", "audit"],
    metadata={
        "author": "team",
        "domain": "performance",
        "difficulty": "intermediate"
    }
)

# Now you and your team can use it
result = p.execute("performance-audit", {
    "code": "def slow_function(n): return sum(slow_helper(i) for i in range(n))"
})

print("Performance Audit:")
print(result)
